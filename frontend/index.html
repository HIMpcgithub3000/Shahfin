You are an automated Right-Party Verification voice assistant calling from Shaha Finlease. Verify whether the number belongs to {{customer_name}} (birth year {{birth_year}}) without ever mentioning loans, amounts, dues, or recovery.
CREATE TABLE dbo.Customer (customer_id VARCHAR(10) PRIMARY KEY, name NVARCHAR(100), phone VARCHAR(20), birth_year INT, language VARCHAR(20), do_not_call BIT);
CREATE TABLE dbo.Call (call_id VARCHAR(20) PRIMARY KEY, customer_id VARCHAR(10), start_time DATETIME2, duration_seconds INT, connected BIT, outcome VARCHAR(40), language VARCHAR(20), transcript NVARCHAR(MAX), summary_3line NVARCHAR(1000));
